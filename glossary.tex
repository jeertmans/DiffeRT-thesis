% Glossary entries

%% B
\newglossaryentry{bindings}{
  name=(\glsxtrshort{api}) bindings,
  description={refer to libraries that allow a programming language to use a library or service written in a different language}
}

%% F
\newglossaryentry{far field}{
  name=far field,
  description={is the region far from an antenna, where the electromagnetic field behaves as a simple radiating wave and the wavefronts can be approximated as locally planar}
}
\newglossaryentry{free space}{
  name=free space,
  description={refers to a perfect vacuum, but in practice is often used to refer to air as they have similar electromagnetic properties at radio frequencies}
}

%% G
\newglossaryentry{geodesy}{
  name=geodesy,
  description={is the science of measuring and understanding the Earth's geometric shape, orientation in space, and gravity field}
}

%% J
\newglossaryentry{jit-compilation}{
  name=just-in-time compilation,
  description={is the process of compiling code upon its first execution, in order to generate optimized machine code for subsequent executions}
}

%% L
\newglossaryentry{line-of-sight}{
  name=line-of-sight,
  description={is the region where a direct path exists between the transmitter and receiver, without any obstacles in between}
}

%% N
\newglossaryentry{near field}{
  name=near field,
  description={is the region close to an antenna, where the electromagnetic field does not behave as a simple radiating wave and is dominated by reactive fields}
}

%% R
\newglossaryentry{rasterization}{
  name=rasterization,
  description={is the process of converting vector graphics into a raster image (pixels or dots), usually for output on a video display or printer}
}

%% S
\newglossaryentry{shadow region}{
  name=shadow region,
  description={is a spatial region where a signal, whether direct or indirect, emitted from a source is blocked by an obstacle}
}

%% W
\newglossaryentry{wavelength}{
  name=wavelength,
  description={is the spatial period of a wave (in \unit{\meter}), defined as the distance over which the wave's shape repeats}
}
\newglossaryentry{wavenumber}{
  name=wavenumber,
  description={is the spatial frequency of a wave (in \unit{\per\meter}), defined as the number of \glspl{wavelength} per unit distance}
}
