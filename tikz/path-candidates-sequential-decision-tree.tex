    tree_leaf/.style={tree_root},
    edge_valid/.style={->, draw=green!80!black, thick, opacity=0.75},

  \draw[edge_invalid] (T3) -- (T32) node[prune_label] {Prune};

  \draw[edge_valid_2] (T31) -- (T31R);
