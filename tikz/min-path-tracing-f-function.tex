\begin{tikzpicture}[
    scale=0.6,
    ray/.style={->,thick,draw=green!80!black,opacity=0.75},
    invalid ray/.style={ray,dashed},
    wall/.style={draw=red, thick},
    normal/.style={->, thick},
  ]
  \tikzset{point/.style={draw,cross out,thick,inner sep=2pt}}
  \tikzstyle{ground}=[fill,draw=none,minimum width=0.3,minimum height=0.6]
  \tikzstyle{intersection-point}=[draw=blue,fill=white,fill opacity=0.5,circle,thick,inner sep=2pt]

  % Fixed Transmit/Receive positions
  \def\TxX{-2.5} \def\TxY{3}
  \def\RxX{2.5}  \def\RxY{3}

  % Wall boundaries
  \def\WallStart{-1.5} \def\WallEnd{1.5}

  % ---------------------------------------------------------
  % Case 1: C = 0.00 + 0.00 (On surface)
  % ---------------------------------------------------------
  \begin{scope}[xshift=0cm]
    \node[point] (Tx) at (\TxX, \TxY) {};
    \node[point] (Rx) at (\RxX, \RxY) {};
    \coordinate (S)  at (0, 0); % Specular point

    % Rays
    \draw[ray] (Tx) -- (S) -- (Rx);

    % Wall & Normal
    \draw[ground,pattern={Lines[angle=-45]}] (\WallStart, 0) rectangle (\WallEnd, -0.1);
    \draw[wall] (\WallStart, 0) -- (\WallEnd, 0);
    \draw[normal] (S) -- +(0, 2) node[above] {$\uvec{n}$};

    % Arcs
    \draw[->,thick] (S) ++(90:1) arc (90:129.74:1) node[midway,above,font=\footnotesize] {$\theta_i$} node[pos=1, left,font=\footnotesize] {\qty{39.74}{\degree}};
    \draw[->,thick]  (S) ++(90:1) arc (90:50.26:1) node[midway,above,font=\footnotesize] {$\theta_r$} node[pos=1, right,font=\footnotesize] {\qty{39.74}{\degree}};

    % Points
    \node[intersection-point] at (S) {};

    % Labels
    \node[anchor=north] at (0,-0.5) {$C = \underbrace{0.00}_{C^{\mathrm{int}}} + \underbrace{0.00}_{C^{\mathrm{obj}}}$};
  \end{scope}

  % ---------------------------------------------------------
  % Case 2: C = 0.00 + 0.07 (Slightly above surface)
  % ---------------------------------------------------------
  \begin{scope}[xshift=7cm]
    \node[point] (Tx) at (\TxX, \TxY) {};
    \node[point] (Rx) at (\RxX, \RxY) {};
    \coordinate (S)  at (0, 0.27); % Specular point shifted up

    % Rays
    \draw[invalid ray] (Tx) -- (S) -- (Rx);

    % Wall & Normal
    \draw[ground,pattern={Lines[angle=-45]}] (\WallStart, 0) rectangle (\WallEnd, -0.1);
    \draw[wall] (\WallStart, 0) -- (\WallEnd, 0);
    \draw[normal] (S) -- +(0, 2) node[above] {$\uvec{n}$};

    % Arcs
    \draw[->, thick] (S) ++(90:1) arc (90:132.41:1) node[midway,above,font=\footnotesize] {$\theta_i$} node[pos=1, left,font=\footnotesize] {\qty{42.41}{\degree}};
    \draw[->, thick]  (S) ++(90:1) arc (90:47.59:1) node[midway,above,font=\footnotesize] {$\theta_r$} node[pos=1, right,font=\footnotesize] {\qty{42.41}{\degree}};

    % Points
    \node[intersection-point] at (S) {};

    % Labels (Fixed at y=-0.5 to align across figures)
    \node[anchor=north] at (0,-0.5) {$C = \underbrace{0.00}_{C^{\mathrm{int}}} + \underbrace{0.07}_{C^{\mathrm{obj}}}$};
  \end{scope}

  % ---------------------------------------------------------
  % Case 3: C = 0.00 + 1.00 (Far above surface)
  % ---------------------------------------------------------
  \begin{scope}[xshift=14cm]
    \node[point] (Tx) at (\TxX, \TxY) {};
    \node[point] (Rx) at (\RxX, \RxY) {};
    \coordinate (S)  at (0, 1.0); % Specular point shifted far up

    % Rays
    \draw[invalid ray] (Tx) -- (S) -- (Rx);

    % Wall & Normal
    \draw[ground,pattern={Lines[angle=-45]}] (\WallStart, 0) rectangle (\WallEnd, -0.1);
    \draw[wall] (\WallStart, 0) -- (\WallEnd, 0);
    \draw[normal] (S) -- +(0, 2) node[above] {$\uvec{n}$};

    % Arcs
    \draw[->, thick] (S) ++(90:1) arc (90:141.43:1) node[midway,above,font=\footnotesize] {$\theta_i$} node[pos=1, left,font=\footnotesize] {\qty{51.43}{\degree}};
    \draw[->, thick]  (S) ++(90:1) arc (90:38.57:1) node[midway,above,font=\footnotesize] {$\theta_r$} node[pos=1, right,font=\footnotesize] {\qty{51.43}{\degree}};

    % Points
    \node[intersection-point] at (S) {};

    % Labels (Fixed at y=-0.5 to align across figures)
    \node[anchor=north] at (0,-0.5) {$C = \underbrace{0.00}_{C^{\mathrm{int}}} + \underbrace{1.00}_{C^{\mathrm{obj}}}$};
  \end{scope}

\end{tikzpicture}
