% Requires:
% \usetikzlibrary{arrows.meta}
\begin{tikzpicture}[
    scale=0.36, transform shape,
    radiating/.style={->, thick,blue},
    reactive/.style={<->, thick,red},
    region/.style={dotted, thick},
    every node/.style={transform shape=false,align=center,font=\footnotesize},
  ]
  \tikzset{>={Latex[scale=0.6]}}

  % Coordinates
  \coordinate (O) at (0,0);

  % Concentric circles
  \draw[region] (O) circle (1.8cm); % Inner Near-Field
  \draw[region] (O) circle (4.0cm); % Outer Near-Field boundary

  % Near-field region label
  \node at (0, 2.7) {Near-field\\region};

  % Far-field region label
  \node[anchor=north] at (0.0, -4.2) {Far-field region};

  % Dimension L (Antenna size / Inner boundary)
  \draw[|<->|,thick] (-150:1.8) -- (30:1.8) node[pos=.75,below,sloped] {$D$};

  \node[circle,draw,fill=red,thick,minimum size=0.3cm] (source) at (O) {};

  % Dimension d_f (Fraunhofer distance)
  \draw[|<->|,thick] (source) -- (-60:4.0) node[pos=.7,below,sloped] {$d_f$};

  \foreach \deg in {-30, 90, 150, 240} {
    \draw[reactive] (\deg:0.6) -- (\deg:1.6);
  }
  \foreach \deg in {-30, 30, 150, 210, 240} {
    \draw[reactive] (\deg:2.0) -- (\deg:3.8);
  }
  \foreach \deg in {0,180,270} {
    \draw[radiating] (\deg:2.0) -- (\deg:3.8);
  }
  \foreach \deg in {45,-45,135,225} {
    \draw[radiating] (\deg:4.2) -- (\deg:5.8);
  }
  \begin{scope}[shift={(-3.3, 4.7)}]
    \draw[radiating] (0, 0.6) -- (1, 0.6) node[right,font=\footnotesize] {Radiating fields};
    \draw[reactive] (0, 0) -- (1, 0) node[right,font=\footnotesize] {Reactive fields};
  \end{scope}

\end{tikzpicture}
