% Requires:
% \usetikzlibrary{patterns}
% \usetikzlibrary{shapes.misc}
% \providecommand{\glsxtrshort}[1]{#1}
\begin{tikzpicture}
  \tikzset{point/.style={draw,circle,thick,inner sep=2pt}}
  \tikzset{tx/.style={point,cross out}}
  \tikzset{rx/.style={point,cross out}}
  \tikzset{invisible/.style={opacity=0.25}} % For invisible labels

  \coordinate (e1) at (-1, 0.5);
  \coordinate (e2) at (0, 0.5);
  \coordinate (e3) at (-1, 1.5);
  \coordinate (e4) at (2, -0.5);
  \coordinate (e5) at (3.5, -0.5);
  \coordinate (e6) at (3.5, 1.0);

  % 1. Background Bounding Box
  \clip (-1.5, -1.2) rectangle (4., 2.8);

  % 2. Shadow Polygons from RX (2, 2)
  % Calculated by casting rays from RX past e3, e2 (for Obj 1) and e6, e4 (for Obj 2)
  \fill[black!10!white] (-3, 1.166) -- (e3) -- (e1) -- (e2) -- (-3, -1.75) -- cycle;
  \fill[black!10!white] (5, 0) -- (e6) -- (e5) -- (e4) -- (2, -2) -- (5, -2) -- cycle;

  % 3. Dashed Line-of-Sight Rays
  \draw[dashed, gray, thick, opacity=0.5] (2, 2) -- (5, 0);       % Ray past e6
  \draw[dashed, gray, thick, opacity=0.5] (2, 2) -- (2, -2);      % Ray past e4
  \draw[dashed, gray, thick, opacity=0.5] (2, 2) -- (-3, 1.166);  % Ray past e3
  \draw[dashed, gray, thick, opacity=0.5] (2, 2) -- (-3, -1.75);  % Ray past e2

  % 4. Nodes (drawn after fill so they sit on top)
  \node[tx, fill=white, label=below:{\glsxtrshort{tx}}] (TX) at (-.3,-.5) {};
  \node[rx, fill=white, label=above:{\glsxtrshort{rx}}] (RX) at (2,2) {};

  % 5. Objects

  % Object 1 (Left): Visible edge is s2 (e2 to e3)
  \fill[pattern=north east lines, thick, pattern color=black] (e1) -- (e2) -- (e3) -- cycle;
  \path[draw=black, thick]
  (e1) edge[invisible] (e2)
  (e2) edge (e3)
  (e3) edge[invisible] (e1);
  \path[text=black]
  (e1) node[point,red,fill=red,invisible,label={[anchor=north east,red,invisible]$e_1$}] {}
  -- (e2) node[midway,below,invisible] {$s_1$} node[point,red,fill=red,label={[anchor=north west,red]$e_2$}] {}
  -- (e3) node[midway,above,sloped] {$s_2$} node[point,red,fill=red,label={[anchor=south,red]$e_3$}] {}
  -- (e1) node[midway,left,invisible] {$s_3$};

  % Object 2 (Right): Visible edge is s6 (e6 to e4)
  \fill[pattern=north east lines, thick, pattern color=black] (e4) -- (e5) -- (e6) -- cycle;
  \path[draw=black, thick]
  (e4) edge[invisible] (e5)
  (e5) edge[invisible] (e6)
  (e6) edge (e4);
  \path[text=black]
  (e4) node[point,red,fill=red,label={[anchor=north east,red,yshift=-0.1cm]$e_4$}] {}
  -- (e5) node[midway,below,invisible] {$s_4$} node[point,red,fill=red,invisible,label={[anchor=north west,red,yshift=-0.1cm,invisible]$e_5$}] {}
  -- (e6) node[midway,right,invisible] {$s_5$} node[point,red,fill=red,label={[anchor=south,red]$e_6$}] {}
  -- cycle node[midway,above,sloped] {$s_6$};
\end{tikzpicture}
