% Requires:
% \usepackage{amsmath}
% \usetikzlibrary{intersections}
% \providecommand{\dual}[1]{\dot{#1}}
\begin{tikzpicture}
  \pgfdeclarelayer{foreground}
  \pgfsetlayers{main,foreground}
  \def\rcRadius{1mm}
  \def\rcShift{0.315*\rcRadius}
  \tikzset{op/.style={draw, thick, fill=gray, fill opacity=0.5, text opacity=1, rounded corners=\rcRadius, minimum width=1.5em, minimum height=1.5em, inner sep=.2em}}
  \tikzset{input/.style={font=\footnotesize}}
  \tikzset{dual input/.style={input,red}}
  \tikzset{constant/.style={input}}
  \tikzset{output/.style={input}}
  \tikzset{dual output/.style={dual input}}
  \tikzset{flow/.style={->,thick}}
  \tikzset{dual flow/.style={->,thick,red,dotted,shorten <=0pt,shorten >=2pt}}
  \tikzset{flow label/.style={above,midway,sloped,font=\scriptsize,fill=white,fill opacity=0.5,text opacity=1,rounded corners=\rcRadius,inner sep=0pt,outer sep=2.5pt}}
  \tikzset{dual label/.style={flow label,below,text=red}}
  \node[op] (square) at (0,0) {$\cdot^2$};
  \node[op,anchor=west,xshift=1.5cm] (exp) at (square.east) {$\exp(\cdot)$};
  \node[op,anchor=north,yshift=-1cm] (cos) at (exp.south) {$\cos(\cdot)$};
  \node[op,anchor=north,yshift=-.3cm] (sin) at (cos.south) {$\sin(\cdot)$};
  \node[op,anchor=west,xshift=2.3cm] (mul1) at (exp.east) {$\times$};
  \path (sin) -- (cos) coordinate[midway] (sin_cos_mid);

  % N.B.: text opacity is 0.5 for zero dual values

  \path (square.west) -- (-.7cm,0) coordinate (x);
  \node[input,anchor=south east] at (x) {$x$};
  \node[dual input,align=left,anchor=north east] at (x) {
    $
    \begin{aligned}
      \dual{x}&=\frac{\partial x}{\partial x} \\
      &= 1
    \end{aligned}$
  };
  \coordinate (y) at (x |- sin_cos_mid);
  \node[input,anchor=south east] at (y) {$y$};
  \node[dual input,align=left,anchor=north east,text opacity=.5] at (y) {
    $
    \begin{aligned}
      \dual{y}&=\frac{\partial y}{\partial x} \\
      &= 0
    \end{aligned}$
  };

  \node[op] (mul2) at (mul1 |- sin_cos_mid) {$\times$};

  \node[op,anchor=west,xshift=2.4cm] (add1) at (mul1.east) {$+$};
  \node[op] (add2) at (add1 |- mul2) {$+$};

  \path (add1) -- (add2) node[midway,constant] (c) {$C$};

  \path (add1.east) -- ++(0.3cm,0) coordinate (f1);
  \node[output,anchor=south west] at (f1) {$z_1 + C$};
  \node[dual output,anchor=north west] at (f1) {$\dual{z}_1=\frac{\partial f_1}{\partial x}$};
  \coordinate (f2) at (f1 |- add2);
  \node[output,anchor=south west] at (f1 |- add2) {$z_2 + C$};
  \node[dual output,anchor=north west] at (f1 |- add2) {$\dual{z}_2=\frac{\partial f_2}{\partial x}$};

  \draw[flow] (x) -- (square);
  \draw[dual flow] (x) -- (square);
  \draw[flow] (square) -- (exp) node[flow label] {$u=x^2$} node[dual label] {$\dual{u}=2\dual{x}x$};
  \draw[dual flow] (square) -- (exp);
  \draw[flow] (exp) -- (mul1) node[flow label] {$v=e^{u}$} coordinate[midway] (exp_mul1_mid);
  \draw[dual flow] (exp) -- (mul1);
  \begin{pgfonlayer}{foreground}
    \path (exp) -- (mul1) node[dual label,below] {$\dual{v}=\dual{u}e^{u}$};
  \end{pgfonlayer}
  \begin{pgfinterruptboundingbox}
    \path [name path=VERT] (exp_mul1_mid) -- (exp_mul1_mid |- sin);
    \path [name path=DIAG_1] (mul2.north west) -- ++(135:5cm);
    \path [name intersections={of=VERT and DIAG_1,by=mid_1}];
    \path [name path=HORI] (cos.east) -- (mul2 |- cos.east);
    \path [name path=DIAG_2] (mul1.south west) -- ++(225:5cm);
    \path [name intersections={of=HORI and DIAG_2,by=mid_2}];
  \end{pgfinterruptboundingbox}
  \draw[flow,rounded corners=\rcRadius] (exp_mul1_mid) -- (mid_1)  -- ([xshift=\rcShift, yshift=-\rcShift]mul2.north west);
  \draw[dual flow,rounded corners=\rcRadius] (exp_mul1_mid) -- (mid_1)  -- ([xshift=\rcShift, yshift=-\rcShift]mul2.north west);

  \draw[flow] (y) -- (cos);
  \draw[dual flow] (y) -- (cos);
  \draw[flow] (y) -- (sin);
  \draw[dual flow] (y) -- (sin);

  \draw[flow,rounded corners=\rcRadius] (cos.east) -- (mid_2) -- ([xshift=\rcShift, yshift=\rcShift]mul1.south west) node[flow label,pos=.45] {$w_1=\cos(y)$} node[dual label,pos=.45,text opacity=.5] {$\dual{w}_1=-\dual{y}\sin(y)$};
  \draw[dual flow,rounded corners=\rcRadius] (cos.east) -- (mid_2) -- ([xshift=\rcShift, yshift=\rcShift]mul1.south west);
  \draw[flow] (sin.east) -- (mul2.west) node[flow label] {$w_2=\sin(y)$} node[dual label, text opacity=.5] {$\dual{w}_2=\dual{y}\cos(y)$};
  \draw[dual flow] (sin.east) -- (mul2.west);

  \draw[flow] (mul1) -- (add1) node[flow label] {$z_1 = w_1 v$} node[dual label] {$\dual{z}_1=\textcolor{red!50}{\dual{w}_1 v} + \dual{v}w_1$};
  \draw[dual flow] (mul1) -- (add1);
  \draw[flow] (mul2) -- (add2) node[flow label] {$z_2 = w_2 v$} node[dual label] {$\dual{z}_2=\textcolor{red!50}{\dual{w}_2 v} + \dual{v}w_2$};
  \draw[dual flow] (mul2) -- (add2);
  \draw[flow] (c) -- (add1);
  %\draw[dual flow] (c) -- (add1);
  \draw[flow] (c) -- (add2);
  %\draw[dual flow] (c) -- (add2);

  \draw[flow] (add1) -- (f1);
  \draw[dual flow] (add1) -- (f1);
  \draw[flow] (add2) -- (f2);
  \draw[dual flow] (add2) -- (f2);
\end{tikzpicture}
